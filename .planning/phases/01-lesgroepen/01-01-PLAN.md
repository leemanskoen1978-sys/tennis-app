Deze fase voegt drie dingen toe aan `lib/types.ts`. De namen liggen hier vast omdat plan 02
(het SQL-schema) er in dezelfde golf op moet aansluiten, kolom voor kolom:

LesGroep — velden, in deze volgorde, met exact deze namen:
  id: string
  name: string
  level: string
  weekday: number            (0-6, zondag = 0, zoals Date#getDay)
  start_hour: number         (0-23)
  start_minute: number       (0-59)
  coach_id?: string
  court_id?: string
  season_start: string       (dagsleutel 'jjjj-mm-dd', zoals Vakantie.van)
  season_end: string
  roster: string[]           (speler-id's, zonder betaler-begrip — een groep heeft er geen)
  archived: boolean
  created_at?: string

Booking krijgt: group_id?: string
Settings krijgt: lesson_duration_minutes?: number

De exports van lib/lesgroepen.ts, met hun signatuur:
  export type GroepBoeking = Pick<Booking, 'id' | 'group_id' | 'start_time' | 'status' | 'participant_ids'>
  export function lesGroepFout(g: Omit<LesGroep, 'id'>): string | null
  export function lessenVanGroep<B extends GroepBoeking>(bookings: B[], groupId: string): B[]
  export function groupBookingsFrom<B extends GroepBoeking>(bookings: B[], groupId: string, now: Date): B[]
  export function komendeLessen<B extends GroepBoeking>(bookings: B[], groupId: string, now: Date): B[]
  export function planRosterChange(group: LesGroep, newRoster: string[], bookings: GroepBoeking[], now: Date): RosterChangePlan
  export function groepSleutel(g: Pick<LesGroep, 'name' | 'weekday' | 'start_hour'>): string
  export function actieveGroepen(groepen: LesGroep[]): LesGroep[]
  export function gearchiveerdeGroepen(groepen: LesGroep[]): LesGroep[]

  export interface RosterChangePlan {
    group: LesGroep;
    bookingPatches: Array<{ id: string; participant_ids: string[] }>;
  }
