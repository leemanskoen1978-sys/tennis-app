export interface CoachTotal {
  coachId: string;
  name: string;
  /** Aantal lessen van deze trainer in de selectie, geannuleerde niet meegerekend. */
  lessons: number;
  /** Wat hij daarvoor krijgt: zijn eigen uurtarief naar rato van de duur. */
  amount: number;
  /** Waar `amount` bewust 0 is omdat er nog geen uurtarief is ingevuld. */
  missingRate: boolean;
}
export function payoutsByCoach(bookings: Booking[], users: User[]): CoachTotal[];
export function countedBookings(bookings: Booking[]): Booking[];   // alles behalve 'cancelled'

export function actieveGroepen(groepen: LesGroep[]): LesGroep[];          // !archived
