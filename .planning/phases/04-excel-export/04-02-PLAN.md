// lib/xlsx.ts
export type XlsxCel =
  | { soort: 'tekst'; waarde: string }
  | { soort: 'getal'; waarde: number }
  | { soort: 'geld'; waarde: number }
  | { soort: 'datum'; waarde: Date };
export interface XlsxBlad {
  naam: string;
  koppen: readonly string[];
  rijen: ReadonlyArray<readonly XlsxCel[]>;
  breedtes?: readonly number[];
}
export function buildWorkbook(bladen: XlsxBlad[]): Uint8Array;   // uit plan 04-01

// lib/lesgever.ts — DE ENIGE plek die 'wie gaf deze les' beantwoordt

export function groupSize(b: GroupBooking): number;

// lib/datetime.ts — uit plan 04-01

  id, player_id, participant_ids?, coach_id, taught_by_id?, court_id,
  start_time /* ISO */, end_time /* ISO */, status, group_id?, attendance?
